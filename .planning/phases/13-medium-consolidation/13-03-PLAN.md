      ```sql
      -- ========================================================================
      -- 020_consolidate_rls.sql — RLS policy end-state, single idempotent source
      -- ========================================================================
      -- Phase 13 QUA-03 / audit M5.
      --
      -- This migration replaces the scattered RLS history (001, 004, 005, 006,
      -- 007, 009, 014, 016 — all annotated SUPERSEDED). It is IDEMPOTENT:
      -- running it twice produces zero errors. Pattern:
      --     DROP POLICY IF EXISTS <name> ON <table>;
      --     CREATE POLICY <name> ON <table> FOR <verb> USING (...);
      --
      -- Helper functions (is_org_member / is_org_admin) are defined or replaced
      -- via CREATE OR REPLACE FUNCTION at the top.
      --
      -- IMPORTANT: The application connects with the `DATABASE_URL` Postgres
      -- role which bypasses RLS (no auth.uid() context). These policies are
      -- defense-in-depth only — see CLAUDE.md `### Authentication Flow` and
      -- `src/server/lib/access.ts`.
      -- ========================================================================

      BEGIN;

      -- 1. HELPER FUNCTIONS ----------------------------------------------------
      CREATE OR REPLACE FUNCTION public.is_org_member(org uuid)
        RETURNS boolean LANGUAGE sql STABLE AS $$
          SELECT EXISTS (
            SELECT 1 FROM public.organization_users
            WHERE organization_id = org AND user_id = auth.uid()
          )
        $$;

      CREATE OR REPLACE FUNCTION public.is_org_admin(org uuid)
        RETURNS boolean LANGUAGE sql STABLE AS $$
          SELECT EXISTS (
            SELECT 1 FROM public.organization_users
            WHERE organization_id = org AND user_id = auth.uid() AND role = 'admin'
          )
        $$;

      -- 2. ENABLE RLS ON EVERY TENANT-SCOPED TABLE -----------------------------
      ALTER TABLE public.users ENABLE ROW LEVEL SECURITY;
      ALTER TABLE public.organizations ENABLE ROW LEVEL SECURITY;
      ALTER TABLE public.organization_users ENABLE ROW LEVEL SECURITY;
      -- ... (one line per table, full list below)

      -- 3. POLICIES — ORG-SCOPED TABLES ----------------------------------------
      -- For each org-scoped table, four policies: SELECT (member), INSERT/UPDATE/DELETE (admin).
      -- Pattern repeated for: domains, credentials, routes, smtp_endpoints, http_endpoints,
      -- address_endpoints, messages, deliveries, webhooks, webhook_requests, track_domains,
      -- suppressions, statistics, templates, system_branding, email_accounts, lead_lists,
      -- leads, campaigns, sequences, sequence_steps, campaign_leads, outreach_emails,
      -- outreach_analytics, outlook_mailboxes.

      DROP POLICY IF EXISTS domains_select_member ON public.domains;
      CREATE POLICY domains_select_member ON public.domains FOR SELECT
        USING (public.is_org_member(organization_id));
      DROP POLICY IF EXISTS domains_insert_admin ON public.domains;
      CREATE POLICY domains_insert_admin ON public.domains FOR INSERT
        WITH CHECK (public.is_org_admin(organization_id));
      DROP POLICY IF EXISTS domains_update_admin ON public.domains;
      CREATE POLICY domains_update_admin ON public.domains FOR UPDATE
        USING (public.is_org_admin(organization_id));
      DROP POLICY IF EXISTS domains_delete_admin ON public.domains;
      CREATE POLICY domains_delete_admin ON public.domains FOR DELETE
        USING (public.is_org_admin(organization_id));
      -- ... repeat for each org-scoped table.

      -- 4. POLICIES — USER-SCOPED TABLES ---------------------------------------
      -- users: own row only.
      DROP POLICY IF EXISTS users_select_self ON public.users;
      CREATE POLICY users_select_self ON public.users FOR SELECT
        USING (id = auth.uid());
      DROP POLICY IF EXISTS users_update_self ON public.users;
      CREATE POLICY users_update_self ON public.users FOR UPDATE
        USING (id = auth.uid());

      -- user_notifications: own rows only.
      DROP POLICY IF EXISTS user_notifications_own ON public.user_notifications;
      CREATE POLICY user_notifications_own ON public.user_notifications FOR ALL
        USING (user_id = auth.uid()) WITH CHECK (user_id = auth.uid());

      -- 5. POLICIES — MAIL TABLES (user-scoped via mailboxes.user_id) ----------
      DROP POLICY IF EXISTS mailboxes_own ON public.mailboxes;
      CREATE POLICY mailboxes_own ON public.mailboxes FOR ALL
        USING (user_id = auth.uid()) WITH CHECK (user_id = auth.uid());

      DROP POLICY IF EXISTS mail_folders_own ON public.mail_folders;
      CREATE POLICY mail_folders_own ON public.mail_folders FOR ALL
        USING (mailbox_id IN (SELECT id FROM public.mailboxes WHERE user_id = auth.uid()))
        WITH CHECK (mailbox_id IN (SELECT id FROM public.mailboxes WHERE user_id = auth.uid()));
      -- ... mail_messages, mail_filters, signatures, contacts identical pattern.

      -- 6. POLICIES — organization_users (special: member can see own membership, admin can manage)
      DROP POLICY IF EXISTS org_users_select_self_or_member ON public.organization_users;
      CREATE POLICY org_users_select_self_or_member ON public.organization_users FOR SELECT
        USING (user_id = auth.uid() OR public.is_org_member(organization_id));
      -- ... INSERT/UPDATE/DELETE for org admin.

      -- 7. POLICIES — organizations (member can SELECT, admin can mutate, owner can mutate)
      DROP POLICY IF EXISTS organizations_select_member ON public.organizations;
      CREATE POLICY organizations_select_member ON public.organizations FOR SELECT
        USING (public.is_org_member(id));
      DROP POLICY IF EXISTS organizations_update_admin ON public.organizations;
      CREATE POLICY organizations_update_admin ON public.organizations FOR UPDATE
        USING (public.is_org_admin(id));
      DROP POLICY IF EXISTS organizations_delete_owner ON public.organizations;
      CREATE POLICY organizations_delete_owner ON public.organizations FOR DELETE
        USING (owner_id = auth.uid());

      COMMIT;
      ```

    Important authoring rules:
      - Every policy MUST be preceded by `DROP POLICY IF EXISTS <name> ON <table>;` for true idempotence. Re-running yields zero errors.
      - Use `CREATE OR REPLACE FUNCTION` (not `CREATE FUNCTION`) for helpers.
      - Wrap in a single `BEGIN; ... COMMIT;` transaction so partial failure rolls back.
      - When iterating: for every table listed in the §3/§4/§5 inventory above, write the four standard policies (or the appropriate pattern). Use migration 016 as the canonical reference for the EXACT policy names and `WITH CHECK`/`USING` shapes — do not invent new names; match what's in 016 so verify-rls-policies recognizes them.
      - Skip server_id / dropped tables (per migration 008). The `servers` table no longer exists.
